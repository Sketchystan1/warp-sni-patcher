cmake_minimum_required(VERSION 3.20)
file(STRINGS "${CMAKE_CURRENT_SOURCE_DIR}/CHANGELOG.md" VERSION_LINE
     REGEX "^## \\[[0-9]+\\.[0-9]+\\.[0-9]+" LIMIT_COUNT 1)
string(REGEX MATCH "[0-9]+\\.[0-9]+\\.[0-9]+" APP_VERSION "${VERSION_LINE}")
string(REGEX MATCH "\\[([^]]+)\\]" VERSION_HEADER "${VERSION_LINE}")
set(PACKAGE_VERSION "${CMAKE_MATCH_1}")
project(warp-sni-patcher VERSION ${APP_VERSION} LANGUAGES CXX)

set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
if(NOT CMAKE_CONFIGURATION_TYPES AND NOT CMAKE_BUILD_TYPE)
    set(CMAKE_BUILD_TYPE MinSizeRel CACHE STRING "Build type" FORCE)
endif()
option(WARP_ALL_FORMATS "Include backends for other operating systems" OFF)
option(WARP_GUI "Build the platform GUI" ON)
option(WARP_STATIC_RUNTIME "Bundle the MSVC runtime for a standalone executable" ON)
option(WARP_CAPSTONE "Use Capstone for Mach-O instruction validation when available" ON)

if(MSVC AND WARP_STATIC_RUNTIME)
    set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
endif()
include(CheckIPOSupported)
check_ipo_supported(RESULT HAVE_IPO OUTPUT IPO_ERROR)
if(HAVE_IPO)
    set(CMAKE_INTERPROCEDURAL_OPTIMIZATION_RELEASE ON)
    set(CMAKE_INTERPROCEDURAL_OPTIMIZATION_MINSIZEREL ON)
endif()

add_library(warp_core STATIC src/patcher.cpp src/service.cpp)
if(WARP_ALL_FORMATS)
    target_compile_definitions(warp_core PRIVATE WARP_ALL_FORMATS)
endif()
if(WIN32 OR WARP_ALL_FORMATS)
    target_sources(warp_core PRIVATE src/formats/pe.cpp)
    target_compile_definitions(warp_core PRIVATE WARP_HAS_PE)
endif()
if((UNIX AND NOT APPLE) OR WARP_ALL_FORMATS)
    target_sources(warp_core PRIVATE src/formats/elf.cpp)
    target_compile_definitions(warp_core PRIVATE WARP_HAS_ELF)
endif()
if(APPLE OR WARP_ALL_FORMATS)
    target_sources(warp_core PRIVATE src/formats/macho.cpp)
    target_compile_definitions(warp_core PRIVATE WARP_HAS_MACHO)
endif()
target_include_directories(warp_core PUBLIC src)
if(WIN32)
    target_compile_definitions(warp_core PUBLIC NOMINMAX WIN32_LEAN_AND_MEAN _CRT_SECURE_NO_WARNINGS)
    target_link_libraries(warp_core PUBLIC advapi32)
else()
    find_package(Threads REQUIRED)
    target_link_libraries(warp_core PUBLIC Threads::Threads)
endif()
if(WARP_CAPSTONE AND (APPLE OR WARP_ALL_FORMATS))
    set(WARP_CAPSTONE_INCLUDE_DIR "" CACHE PATH "Explicit Capstone include directory for cross builds")
    set(WARP_CAPSTONE_LIBRARY "" CACHE FILEPATH "Explicit Capstone library for cross builds")
    if(WARP_CAPSTONE_INCLUDE_DIR AND WARP_CAPSTONE_LIBRARY)
        if(NOT EXISTS "${WARP_CAPSTONE_INCLUDE_DIR}/capstone/capstone.h" OR NOT EXISTS "${WARP_CAPSTONE_LIBRARY}")
            message(FATAL_ERROR "The supplied Capstone headers or library do not exist")
        endif()
        target_include_directories(warp_core PRIVATE "${WARP_CAPSTONE_INCLUDE_DIR}")
        target_link_libraries(warp_core PRIVATE "${WARP_CAPSTONE_LIBRARY}")
        target_compile_definitions(warp_core PUBLIC WARP_HAS_CAPSTONE)
    elseif(WARP_CAPSTONE_INCLUDE_DIR OR WARP_CAPSTONE_LIBRARY)
        message(FATAL_ERROR "Set both WARP_CAPSTONE_INCLUDE_DIR and WARP_CAPSTONE_LIBRARY")
    else()
        find_package(PkgConfig QUIET)
        if(PkgConfig_FOUND)
            pkg_check_modules(CAPSTONE QUIET IMPORTED_TARGET capstone)
        endif()
        if(CAPSTONE_FOUND)
            target_compile_definitions(warp_core PUBLIC WARP_HAS_CAPSTONE)
            if(APPLE)
                find_file(CAPSTONE_ARCHIVE NAMES libcapstone.a PATHS ${CAPSTONE_LIBRARY_DIRS} NO_DEFAULT_PATH)
            endif()
            if(CAPSTONE_ARCHIVE)
                target_include_directories(warp_core PRIVATE ${CAPSTONE_INCLUDE_DIRS})
                target_link_libraries(warp_core PRIVATE ${CAPSTONE_ARCHIVE})
            else()
                target_link_libraries(warp_core PRIVATE PkgConfig::CAPSTONE)
            endif()
        elseif(APPLE)
            message(FATAL_ERROR "Install Capstone and pkg-config, or explicitly set WARP_CAPSTONE=OFF")
        endif()
    endif()
endif()

add_executable(warp-sni-patcher src/main.cpp)
target_link_libraries(warp-sni-patcher PRIVATE warp_core)
if(WIN32)
    configure_file(resources/version.h.in generated/version.h @ONLY)
    target_sources(warp-sni-patcher PRIVATE resources/app.rc)
    target_include_directories(warp-sni-patcher PRIVATE resources "${CMAKE_CURRENT_BINARY_DIR}/generated")
    if(WARP_GUI)
        set_target_properties(warp-sni-patcher PROPERTIES WIN32_EXECUTABLE TRUE)
        target_sources(warp-sni-patcher PRIVATE src/gui/win32_gui.cpp)
        target_link_libraries(warp-sni-patcher PRIVATE user32 gdi32 comctl32 dwmapi uxtheme shell32)
    endif()
elseif(APPLE AND WARP_GUI)
    enable_language(OBJCXX)
    set_target_properties(warp-sni-patcher PROPERTIES OBJCXX_STANDARD 17 OBJCXX_STANDARD_REQUIRED ON)
    target_compile_options(warp-sni-patcher PRIVATE "$<$<COMPILE_LANGUAGE:OBJCXX>:-fobjc-arc>")
    target_sources(warp-sni-patcher PRIVATE src/gui/macos_gui.mm)
    target_link_libraries(warp-sni-patcher PRIVATE "-framework Cocoa")
elseif(WARP_GUI)
    find_program(FLTK_CONFIG NAMES fltk-config REQUIRED)
    execute_process(COMMAND "${FLTK_CONFIG}" --includedir OUTPUT_VARIABLE FLTK_INCLUDE_DIR
        OUTPUT_STRIP_TRAILING_WHITESPACE COMMAND_ERROR_IS_FATAL ANY)
    execute_process(COMMAND "${FLTK_CONFIG}" --ldstaticflags OUTPUT_VARIABLE FLTK_STATIC_FLAGS
        OUTPUT_STRIP_TRAILING_WHITESPACE COMMAND_ERROR_IS_FATAL ANY)
    if(NOT FLTK_STATIC_FLAGS MATCHES "libfltk\\.a")
        message(FATAL_ERROR "Install the FLTK development package with its static libfltk.a archive")
    endif()
    separate_arguments(FLTK_STATIC_LIBRARIES UNIX_COMMAND "${FLTK_STATIC_FLAGS}")
    target_sources(warp-sni-patcher PRIVATE src/gui/linux_gui.cpp)
    target_include_directories(warp-sni-patcher PRIVATE ${FLTK_INCLUDE_DIR})
    target_compile_definitions(warp-sni-patcher PRIVATE WARP_HAS_FLTK)
    target_link_libraries(warp-sni-patcher PRIVATE ${FLTK_STATIC_LIBRARIES} ${CMAKE_DL_LIBS})
endif()
if(NOT WARP_GUI)
    target_compile_definitions(warp-sni-patcher PRIVATE WARP_NO_GUI)
endif()

if(MSVC)
    foreach(target warp_core warp-sni-patcher)
        target_compile_options(${target} PRIVATE /utf-8 /W3 /Gy /Gw /guard:cf)
    endforeach()
    target_link_options(warp-sni-patcher PRIVATE /OPT:REF /OPT:ICF /INCREMENTAL:NO
        /DYNAMICBASE /NXCOMPAT /HIGHENTROPYVA /guard:cf /RELEASE
        "/MANIFESTUAC:level='asInvoker' uiAccess='false'")
else()
    foreach(target warp_core warp-sni-patcher)
        target_compile_options(${target} PRIVATE -Wall -Wextra -ffunction-sections -fdata-sections)
    endforeach()
    if(APPLE)
        target_link_options(warp-sni-patcher PRIVATE -Wl,-dead_strip,-S,-x,-adhoc_codesign)
    else()
        target_link_options(warp-sni-patcher PRIVATE -Wl,--gc-sections)
    endif()
endif()

include(CTest)
if(BUILD_TESTING)
    add_executable(test_patcher tests/test_patcher.cpp)
    target_link_libraries(test_patcher PRIVATE warp_core)
    if(MSVC)
        target_compile_options(test_patcher PRIVATE /utf-8)
    endif()
    add_test(NAME PatcherTests COMMAND test_patcher)
    find_package(Python3 COMPONENTS Interpreter QUIET)
    if(Python3_Interpreter_FOUND)
        add_test(NAME CliTests COMMAND ${CMAKE_COMMAND} -E env
            "WARP_PATCHER_EXE=$<TARGET_FILE:warp-sni-patcher>"
            ${Python3_EXECUTABLE} "${CMAKE_CURRENT_SOURCE_DIR}/scripts/test_cli.py" -v)
        add_test(NAME ReleaseTests COMMAND ${Python3_EXECUTABLE}
            "${CMAKE_CURRENT_SOURCE_DIR}/scripts/test_release.py" -v)
    endif()
endif()

install(TARGETS warp-sni-patcher RUNTIME DESTINATION .)
string(TOLOWER "${CMAKE_SYSTEM_NAME}" PACKAGE_PLATFORM)
if(APPLE)
    set(PACKAGE_PLATFORM macos)
endif()
set(CPACK_PACKAGE_FILE_NAME "warp-sni-patcher-v${PACKAGE_VERSION}-${PACKAGE_PLATFORM}")
set(CPACK_INCLUDE_TOPLEVEL_DIRECTORY OFF)
# Darwin's linker signs the final image. Post-link stripping would invalidate it.
set(CPACK_STRIP_FILES TRUE)
if(APPLE)
    set(CPACK_STRIP_FILES FALSE)
endif()
set(CPACK_GENERATOR ZIP)
if(UNIX AND NOT APPLE)
    set(CPACK_GENERATOR TGZ)
endif()
include(CPack)
